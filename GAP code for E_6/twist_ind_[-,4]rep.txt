######## ISOTROPY OF INDUCTION OF [-,4]-REP #########

W:=CoxeterGroup("E",6);;
C:=ComplexReflectionGroup(3,1,1);;
G:=DirectProduct(W,C);;
s1:=Reflections(W)[1];;
s2:=Reflections(W)[2];;
s3:=Reflections(W)[3];;
s4:=Reflections(W)[4];;
s5:=Reflections(W)[5];;
s6:=Reflections(W)[6];;
s0:=Reflections(W)[36];;
x:=(73,74,75);;


omega:=();;

for w in Elements(W) do
if 2^w=5 and 5^w=3 and 3^w=2 and 4^w=4 then
omega:=w;
fi;
od;

H:=Subgroup(G,[s2,s3,s4,s5,omega*x]);;
W_M:=ReflectionSubgroup(W,[2,3,4,5]);;
rcosets:=RightCosets(W,W_M);;
z:=List(rcosets, c -> Elements(c)[1]);;

s1_perm:=List([1..270], i -> First([1..270], j -> z[i]*s1 in H*z[j]));;
s2_perm:=List([1..270], i -> First([1..270], j -> z[i]*s2 in H*z[j]));;
s3_perm:=List([1..270], i -> First([1..270], j -> z[i]*s3 in H*z[j]));;
s4_perm:=List([1..270], i -> First([1..270], j -> z[i]*s4 in H*z[j]));;
s5_perm:=List([1..270], i -> First([1..270], j -> z[i]*s5 in H*z[j]));;
s6_perm:=List([1..270], i -> First([1..270], j -> z[i]*s6 in H*z[j]));;
s0_perm:=List([1..270], i -> First([1..270], j -> z[i]*s0 in H*z[j]));;

x_perm:=List([1..270], i -> First([1..270], j -> z[i]*x in H*z[j]));;

s1_act:=NullMat(270);;
s2_act:=NullMat(270);;
s3_act:=NullMat(270);;
s4_act:=NullMat(270);;
s5_act:=NullMat(270);;
s6_act:=NullMat(270);;
s0_act:=NullMat(270);;
x_act:=NullMat(270);;


for i in [1..270] do
s1_act[i][s1_perm[i]]:=1;
s2_act[i][s2_perm[i]]:=1;
s3_act[i][s3_perm[i]]:=1;
s4_act[i][s4_perm[i]]:=1;
s5_act[i][s5_perm[i]]:=1;
s6_act[i][s6_perm[i]]:=1;
s0_act[i][s0_perm[i]]:=1;
od;

act:=[s1_act,s2_act,s3_act,s4_act,s5_act,s6_act,s0_act];;



#_________________________________________________________________
##### Next we compute the action of Ta0

s:=[1,1,0,2,0,1];;

eval:=[];

for i in [1..270] do
w:=z[i];
walpha1:=W.roots[1^w];
walpha2:=W.roots[2^w];
walpha3:=W.roots[3^w];
walpha4:=W.roots[4^w];
walpha5:=W.roots[5^w];
walpha6:=W.roots[6^w];
ws:=List([1..6], i-> walpha1[i]+walpha2[i]+2*walpha4[i]+walpha6[i]);
Add(eval,E(3)^ws[2]);
od;

Ta0:=NullMat(270);;
for i in [1..270] do
Ta0[i][i]:=eval[i];
od;

act[7]:=s0_act*Ta0;;



#_________________________________________________________________
###### Determine the action of x, most challenging
x_reps_in_WM:=List(x_reps, w -> w*omega^-1*x^-1);;
x_words:=List(x_reps_in_WM, w -> CoxeterWord(W_M, w));;

x_act:=NullMat(270);;
for i in [1..270] do
x_act[i][x_perm[i]]:=1;
od;

E_plus := NullspaceMat(x_act - IdentityMat(270));;
E_theta := NullspaceMat(x_act - E(3)*IdentityMat(270));;
E_theta2 := NullspaceMat(x_act - E(3)^2*IdentityMat(270));;
P:=Concatenation(E_plus, E_theta, E_theta2);;

act_block:=List(act, A -> P*A*P^-1);;
act_plus:=List(act_block, B -> B{[1..90]}{[1..90]});;
act_theta:=List(act_block, B -> B{[91..180]}{[91..180]});;
act_theta2:=List(act_block, B -> B{[181..270]}{[181..270]});;




#______________________________________________________________________
#______________________________________________________________________
###### Determine the action of the full Weyl group W=<1,2,3,4,5,6>

classesW:=ConjugacyClasses(W);;
sizesW:=List(classesW, c-> Size(c));
wordsW:=List(classesW, c-> CoxeterWord(W,Elements(c)[1]));
signW:=List(wordsW, s -> (-1)^Length(s));;


matricesW_plus:=[];;
matricesW_theta:=[];;
matricesW_theta2:=[];;

for k in [1..25] do
matrix_plus:=IdentityMat(90);
matrix_theta:=IdentityMat(90);
matrix_theta2:=IdentityMat(90);
for j in wordsW[k] do
matrix_plus:=matrix_plus*act_plus[j];
matrix_theta:=matrix_theta*act_theta[j];
matrix_theta2:=matrix_theta2*act_theta2[j];
od;
Add(matricesW_plus, matrix_plus);
Add(matricesW_theta, matrix_theta);
Add(matricesW_theta2, matrix_theta2);
od;

chiW_plus:=List([1..25], i -> Trace(matricesW_plus[i])*signW[i]);
chiW_theta:=List([1..25], i -> Trace(matricesW_theta[i])*signW[i]);
chiW_theta2:=List([1..25], i -> Trace(matricesW_theta2[i])*signW[i]);

for k in [1..25] do
rep:=Irr(CharTable(W))[k].values;
total_plus:=0;
total_theta:=0;
total_theta2:=0;
for j in [1..25] do
total_plus:=total_plus+rep[j]*chiW_plus[j]*sizesW[j];
total_theta:=total_theta+rep[j]*chiW_theta[j]*sizesW[j];
total_theta2:=total_theta2+rep[j]*chiW_theta2[j]*sizesW[j];
od;
Print(total_plus/51840, "   ", total_theta/51840, "   ", total_theta2/51840);
Print("   ", CharTable(W).irredinfo[k].charname, "\n");
od;
