W:=CoxeterGroup("A",3);;
C:=ComplexReflectionGroup(2,1,1);;
G:=DirectProduct(W,C);;
s1:=Reflections(W)[1];;
s2:=Reflections(W)[2];;
s3:=Reflections(W)[3];;
s0:=Reflections(W)[6];;
x:=(13,14);;


omega:=();;

for w in Elements(W) do
if 1^w=2 and 2^w=3 and 3^w=12 and 12^w=1 then
omega:=w^2;
fi;
od;

omega;

H:=Subgroup(G,[s1,s3,omega*x]);
W_M:=ReflectionSubgroup(W,[1,3]);
rcosets:=RightCosets(W,W_M);
z:=List(rcosets, c -> Elements(c)[1]);

s1_perm:=List([1..6], i -> First([1..6], j -> z[i]*s1 in H*z[j]));
s2_perm:=List([1..6], i -> First([1..6], j -> z[i]*s2 in H*z[j]));
s3_perm:=List([1..6], i -> First([1..6], j -> z[i]*s3 in H*z[j]));
s0_perm:=List([1..6], i -> First([1..6], j -> z[i]*s0 in H*z[j]));

x_perm:=List([1..6], i -> First([1..6], j -> z[i]*x in H*z[j]));

s1_act:=NullMat(6);;
s2_act:=NullMat(6);;
s3_act:=NullMat(6);;
s0_act:=NullMat(6);;
x_act:=NullMat(6);;


for i in [1..6] do
s1_act[i][s1_perm[i]]:=1;
s2_act[i][s2_perm[i]]:=1;
s3_act[i][s3_perm[i]]:=1;
s0_act[i][s0_perm[i]]:=1;
od;

act:=[s1_act,s2_act,s3_act,s0_act];;



#_________________________________________________________________
##### Next we compute the action of Ta0

eval:=[];

for i in [1..6] do
w:=z[i];
walpha0:=W.roots[12^(w^-1)];
Add(eval,(-1)^walpha0[2]);
od;


Ta0:=NullMat(6);;
for i in [1..6] do
Ta0[i][i]:=eval[i];
od;

act[4]:=s0_act*Ta0;



#_________________________________________________________________
###### Determine the action of x, most challenging

x_act:=NullMat(6);;
for i in [1..6] do
x_act[i][x_perm[i]]:=1;
od;

E_p := NullspaceMat(x_act - IdentityMat(6));
E_m := NullspaceMat(x_act + IdentityMat(6));
P:=Concatenation(E_p, E_m);

act_block:=List(act, A -> P*A*P^-1);
act_p:=List(act_block, B -> B{[1..3]}{[1..3]});
act_m:=List(act_block, B -> B{[4..6]}{[4..6]});






#______________________________________________________________________
#______________________________________________________________________
###### Determine the action of the full Weyl group W=<1,2,3>

classesW:=ConjugacyClasses(W);
sizesW:=List(classesW, c-> Size(c));
wordsW:=List(classesW, c-> CoxeterWord(W,Elements(c)[1]));
signW:=List(wordsW, s -> (-1)^Length(s));


matricesW_p:=[];;
matricesW_m:=[];;

for k in [1..5] do
matrix_p:=IdentityMat(3);
matrix_m:=IdentityMat(3);
for j in wordsW[k] do
matrix_p:=matrix_p*act_p[j];
matrix_m:=matrix_m*act_m[j];
od;
Add(matricesW_p, matrix_p);
Add(matricesW_m, matrix_m);
od;

chiW_p:=List([1..5], i -> Trace(matricesW_p[i])*signW[i]);
chiW_m:=List([1..5], i -> Trace(matricesW_m[i])*signW[i]);

for k in [1..5] do
rep:=Irr(CharTable(W))[k].values;
total_p:=0;
total_m:=0;
for j in [1..5] do
total_p:=total_p+rep[j]*chiW_p[j]*sizesW[j];
total_m:=total_m+rep[j]*chiW_m[j]*sizesW[j];
od;
Print(total_p/24, "   ", total_m/24);
Print("   ", CharTable(W).irredinfo[k].charname, "\n");
od;
